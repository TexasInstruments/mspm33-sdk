59 /*!****************************************************************************
60  * @file ml-kem.h
61  * @brief Module lattice key encapsulation mechanism (ML-KEM)
62  *
63  * This module implements the FIPS 203 standard for a post-quantum resistant
64  * key encapsulation mechanism (Crystals: Kyber) , providing a KEM
65  * implementation with the following properties:
66  * - IND-CPA secure (indistinguishability under a chosen plaintext attack)
67  * - IND-CCA secure (indistinguishability under a chosen ciphertext attack)
68  *
69  ******************************************************************************
70  */
