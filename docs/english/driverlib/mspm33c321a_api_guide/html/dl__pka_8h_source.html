33  * @file dl_pka.h
34  * @brief PKA Driver Library
35  * @defgroup PKA
36  *
37  * @anchor ti_dl_dl_pka_Overview
38  * # Overview
39  *
40  * The PKA Driver Library allows full configuration of the MSPM33 PKA
41  * (Public Key Accelerator) module.
42  * The Public Key Accelerator (PKA) peripheral provides hardware acceleration
43  * for cryptographic operations including big number arithmetic, elliptic curve
44  * cryptography (ECC), and modular exponentiation. It supports multiple standard
45  * curves including NIST P-curves (P-224, P-256, P-384, P-521), Brainpool curves
46  * (P-256R1, P-384R1, P-512R1), and Curve25519 for efficient public key operations.
47  *
48  * <hr>
49  ******************************************************************************
50  */
