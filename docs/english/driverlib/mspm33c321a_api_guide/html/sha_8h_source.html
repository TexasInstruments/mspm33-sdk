59 /*!****************************************************************************
60  * @file sha.h
61  * @brief SHA secure hash algorithm implementation
62  *
63  * This hash module implements the SHA-224, SHA-256, SHA-384, and SHA-512
64  * algorithms according to FIPS 180-4, as well as SHA3-224, SHA3-256,
65  * SHA3-384, and SHA3-512 variants from FIPS 202.
66  *
67  * XOF functions SHAKE128 and SHAKE256 are implemented from FIPS 202.
68  *
69  * SHA-224/256 integration notes:
70  * 1. A maximum of 512MiB may be hashed by this implementation. (SHA2
71  * standard supports 2 million Terabytes for 224/256)
72  * 2. Input data may be 8-bit aligned. However, output digest must be
73  * 32-bit aligned.
74  * 3. Intermediate values, input data, and final digest value may be
75  * left behind on the stack.
76  *
77  ******************************************************************************
78  */
