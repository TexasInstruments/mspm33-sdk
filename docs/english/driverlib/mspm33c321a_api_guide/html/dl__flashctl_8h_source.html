1078 __STATIC_INLINE uint32_t DL_FlashCTL_getFlashSectorNumberInBank(
1079  FLASHCTL_Regs *flashctl, uint32_t addr)
1080 {
1081  uint8_t numBanks = DL_FactoryRegion_getNumBanks();
1082  uint32_t mainFlashSize = DL_FactoryRegion_getMAINFlashSize();
1083  uint32_t sector, bankSectors, sectorInBank;
1084 
1085  /* Current sector over the whole memory map */
1086  sector = DL_FlashCTL_getFlashSectorNumber(flashctl, addr);
1087  if (numBanks > 1U) {
1088  /* Number of sectors per bank, considered max sector
1089  * count for a bank-adjusted sector. Assume banks are
1090  * evenly distributed */
1091  bankSectors = (mainFlashSize / (2U * (uint32_t) (numBanks)));
1092  /* We will not assume that the maximum number of bank sectors is a
1093  * multiple of 2. Which does mean incurring a modulo operation. */
1094  sectorInBank = sector % bankSectors;
1095  } else {
1096  sectorInBank = sector;
1097  }
1098 
1099  return sectorInBank;
1100 }
