1082 __STATIC_INLINE uint32_t DL_FlashCTL_getFlashSectorNumberInBank(
1083  FLASHCTL_Regs *flashctl, uint32_t addr)
1084 {
1085  uint8_t numBanks = DL_FactoryRegion_getNumBanks();
1086  uint32_t mainFlashSize = DL_FactoryRegion_getMAINFlashSize();
1087  uint32_t sector, bankSectors, sectorInBank;
1088 
1089  /* Current sector over the whole memory map */
1090  sector = DL_FlashCTL_getFlashSectorNumber(flashctl, addr);
1091  if (numBanks > 1U) {
1092  /* Number of sectors per bank, considered max sector
1093  * count for a bank-adjusted sector. Assume banks are
1094  * evenly distributed */
1095  bankSectors = (mainFlashSize / (2U * (uint32_t) (numBanks)));
1096  /* We will not assume that the maximum number of bank sectors is a
1097  * multiple of 2. Which does mean incurring a modulo operation. */
1098  sectorInBank = sector % bankSectors;
1099  } else {
1100  sectorInBank = sector;
1101  }
1102 
1103  return sectorInBank;
1104 }
